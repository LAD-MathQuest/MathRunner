%------------------------------------------------------------------------------%
\begin{frame}{As Funções de Cada Margem}
\begin{itemize}[<+->]
  \item Cada margem é definida por uma função
  \item Essas funções dependem do deslocamento $x$ medido em porcentagem da tela
  \item As margens não podem sair da tela, as funções precisam estar entre 0 e 100
  \item A margem mínima (esquerda ou de baixo) padrão é
  \[
    \text{Mínimo}(x) = 35
  \]
  \item A margem máxima (direita ou de cima) padrão é
  \[
    \text{Máximo}(x) = 65
  \]
\end{itemize}
\end{frame}

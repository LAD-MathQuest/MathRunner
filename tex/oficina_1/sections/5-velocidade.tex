%------------------------------------------------------------------------------%
\begin{frame}{A Função Velocidade}
\begin{itemize}[<+->]
  \item A velocidade do jogo é controlada por uma função
  \item Essa função depende do tempo $t$ em segundos
  \item A velocidade não pode ser negativa
  \item A função padrão é
  \[
    v(t) = 25 + 2t
  \]
\end{itemize}
\end{frame}

\section{Introdução}

%------------------------------------------------------------------------------%
\begin{frame}{Projeto MathRunner}
  \begin{enumerate}[<+->]
    \setlength\itemsep{1em}
    \item Ilustrar como a Matemática pode ser usada na criação de um jogo
    \item Explorar o conceito de função matemática
    \item Criar um jogo sem conhecimentos de programação
  \end{enumerate}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}[t]{Jogando}
\begin{enumerate}[<+->]
  \item Execute o programa \code{MathRunner}
  \item Clique em \code{Abrir}
  \item Selecione o jogo \code{racing}
  \item Clique em executar o jogo
  \item Use a tecla \;\texttt{Q}\; para sair do jogo
\end{enumerate}
